import unittest
import json
import subprocess
import sys

class TestAuditCLI(unittest.TestCase):
    def run_cli(self, data):
        """Helper to run the CLI with specific stdin JSON data."""
        input_str = json.dumps(data)
        process = subprocess.run(
            [sys.executable, "solution.py"],
            input=input_str,
            capture_output=True,
            text=True,
            timeout=5
        )
        return process

    def test_success_contract(self):
        # AC7: Success exit zero, print one JSON object
        data = {"window": [0, 10], "threshold": 1, "intervals": [[1, 2]]}
        res = self.run_cli(data)
        self.assertEqual(res.returncode, 0)
        try:
            output = json.loads(res.stdout)
            self.assertIn("covered", output)
            self.assertIn("peak", output)
            self.assertIn("first_peak", output)
            self.assertIn("alerts", output)
        except json.JSONDecodeError:
            self.fail("Stdout was not valid JSON")

        # AC7: Malformed JSON exits nonzero and produce no stdout
        process = subprocess.run(
            [sys.executable, "solution.py"],
            input="{malformed json",
            capture_output=True,
            text=True,
            timeout=5
        )
        self.assertNotEqual(process.returncode, 0)
        self.assertEqual(process.stdout, "")

    def test_invalid_input_types(self):
        # AC1: Booleans and floats do not qualify as integers
        invalid_cases = [
            {"window": [0.5, 10], "threshold": 1, "intervals": [[1, 2]]},
            {"window": [0, 10], "threshold": 1.2, "intervals": [[1, 2]]},
            {"window": [True, 10], "threshold": 1, "intervals": [[1, 2]]},
            {"window": [0, 10], "threshold": 1, "intervals": [[1.0, 2]]},
            {"window": [0, 10], "threshold": 1, "intervals": [[1, False]]},
            {"window": [0, 10], "threshold": 0, "intervals": [[1, 2]]}, # threshold must be positive
            {"window": [0, 10], "threshold": -1, "intervals": [[1, 2]]}, # threshold must be positive
        ]
        for case in invalid_cases:
            res = self.run_cli(case)
            self.assertNotEqual(res.returncode, 0, f"Failed on case: {case}")
            self.assertEqual(res.stdout, "")

    def test_invalid_range_logic(self):
        # AC1: start < end
        invalid_cases = [
            {"window": [10, 0], "threshold": 1, "intervals": [[1, 2]]},
            {"window": [5, 5], "threshold": 1, "intervals": [[1, 2]]},
            {"window": [0, 10], "threshold": 1, "intervals": [[2, 1]]},
            {"window": [0, 10], "threshold": 1, "intervals": [[2, 2]]},
        ]
        for case in invalid_cases:
            res = self.run_cli(case)
            self.assertNotEqual(res.returncode, 0, f"Failed on case: {case}")
            self.assertEqual(res.stdout, "")

    def test_window_intersection(self):
        # AC2: Intersect each reservation with the window
        data = {
            "window": [0, 10],
            "threshold": 1,
            "intervals": [[-5, 2], [8, 15], [11, 12]]
        }
        # clipped: [0, 2], [8, 10], [11, 12] (this one is outside window, effectively empty)
        res = self.run_cli(data)
        self.assertEqual(res.returncode, 0)
        out = json.loads(res.stdout)
        # covered: [0, 2) is 2, [8, 10) is 2. Total 4.
        self.assertEqual(out["covered"], 4)
        self.assertEqual(out["peak"], 1)
        self.assertEqual(out["first_peak"], 0)
        self.assertEqual(out["alerts"], [[0, 2], [8, 10]])

        # AC2: Validate EVERY supplied reservation, even those outside the window
        data_invalid_outside = {
            "window": [0, 10],
            "threshold": 1,
            "intervals": [[11, 10]] # Invalid range logic
        }
        res_inv = self.run_cli(data_invalid_outside)
        self.assertNotEqual(res_inv.returncode, 0)

    def test_duplicate_intervals(self):
        # AC3: Count each input row separately, including duplicates
        data = {
            "window": [0, 10],
            "threshold": 2,
            "intervals": [[1, 5], [1, 5]]
        }
        res = self.run_cli(data)
        self.assertEqual(res.returncode, 0)
        out = json.loads(res.stdout)
        self.assertEqual(out["peak"], 2)
        self.assertEqual(out["covered"], 4)
        self.assertEqual(out["alerts"], [[1, 5]])

    def test_covered_duration(self):
        # AC4: total duration inside window where >= 1 res is active
        data = {
            "window": [0, 10],
            "threshold": 1,
            "intervals": [[1, 3], [2, 4], [6, 8]]
        }
        # Occupied: [1, 4) and [6, 8). Duration: 3 + 2 = 5
        res = self.run_cli(data)
        out = json.loads(res.stdout)
        self.assertEqual(out["covered"], 5)

    def test_peak_and_first_peak(self):
        # AC5: Max simultaneous and earliest time
        data = {
            "window": [0, 10],
            "threshold": 1,
            "intervals": [[1, 5], [2, 6], [3, 7], [8, 9]]
        }
        # Timeline:
        # 1: count 1
        # 2: count 2
        # 3: count 3 (Peak reached here)
        # 5: count 2
        # 6: count 1
        # 7: count 0
        # 8: count 1
        # 9: count 0
        res = self.run_cli(data)
        out = json.loads(res.stdout)
        self.assertEqual(out["peak"], 3)
        self.assertEqual(out["first_peak"], 3)

        # Case peak 0
        data_empty = {"window": [0, 10], "threshold": 1, "intervals": []}
        res_empty = self.run_cli(data_empty)
        out_empty = json.loads(res_empty.stdout)
        self.assertEqual(out_empty["peak"], 0)
        self.assertIsNone(out_empty["first_peak"])

    def test_alerts_merging(self):
        # AC6: maximal half-open ranges where occupancy >= threshold
        # Merge adjacent qualifying spans even when counts differ
        data = {
            "window": [0, 20],
            "threshold": 2,
            "intervals": [[1, 5], [2, 6], [4, 8], [7, 9], [12, 15], [13, 16]]
        }
        # Timeline:
        # [1, 2): 1
        # [2, 4): 2 (Qualifies)
        # [4, 5): 3 (Qualifies) -> Merge [2, 5)
        # [5, 6): 2 (Qualifies) -> Merge [2, 6)
        # [6, 7): 1
        # [7, 8): 2 (Qualifies)
        # [8, 9): 1
        # [12, 13): 1
        # [13, 15): 2 (Qualifies)
        # [15, 16): 1
        # Alerts: [[2, 6], [7, 8], [13, 15]]
        res = self.run_cli(data)
        out = json.loads(res.stdout)
        self.assertEqual(out["alerts"], [[2, 6], [7, 8], [13, 15]])

    def test_negative_coordinates(self):
        # Endpoints may be negative
        data = {
            "window": [-10, 0],
            "threshold": 1,
            "intervals": [[-15, -5], [-6, -2]]
        }
        # Clipped: [-10, -5], [-6, -2]
        # Combined range: [-10, -2). Duration: 8
        # Peak: 2 at -6
        res = self.run_cli(data)
        out = json.loads(res.stdout)
        self.assertEqual(out["covered"], 8)
        self.assertEqual(out["peak"], 2)
        self.assertEqual(out["first_peak"], -6)
        self.assertEqual(out["alerts"], [[-10, -2]])

if __name__ == "__main__":
    unittest.main()
